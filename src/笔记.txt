dotnet publish SCAssistant.UnoApp.csproj -c Release -f net10.0-ios -r ios-arm64 -p:CodesignKey= -p:CodesignProvision= -p:EnableCodeSigning=false


dotnet publish SCAssistant.UnoApp.csproj -c Release -f net10.0-ios -r ios-arm64 -p:_CanBundleWithUntrustedSource=true -p:CodesignKey="-" -p:CodesignProvision= -p:BuildIpa=true



# 进入 publish 目录
cd bin\Release\net10.0-ios\ios-arm64\publish

# 创建 Payload 目录
mkdir Payload

# 把 .app 移进去
Move-Item SCAssistant.UnoApp.app Payload\

# 压缩成 IPA
Compress-Archive -Path Payload -DestinationPath SCAssistant.UnoApp.zip
Rename-Item SCAssistant.UnoApp.zip SCAssistant.UnoApp.ipa
